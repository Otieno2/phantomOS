12:# Try to bind a file URI from the host into the chroot
13:# - If the URI does not use the scheme 'file' -> does nothing
14:# - If the URI does not point to an absolute path -> does nothing
15:#
16:# $1 = directory of the chroot
17:# $2 = URI of scheme file (file://localhost/absolute-path or file:///absolute-path)
18:Chroot_try_bind_path ()
19:{
20:	local CHROOT
21:	local BIND_SRC
22:	local BIND_DEST
23:	CHROOT="$(readlink -f ${1})"
24:	if echo "${2}" | grep -E -q '^file://(localhost)?/'
25:	then
26:		BIND_SRC="$(readlink -f $(echo ${2} | sed --regexp-extended -e 's|^file://(localhost)?||'))"
27:
28:		BIND_DEST=${CHROOT}${BIND_SRC}
29:		if [ ! -d "${BIND_DEST}" -o \
30:			-z "$(cat /proc/mounts | awk -vdir="${BIND_DEST}" '$2 ~ dir { print $2}')" ]
31:		then
32:			Echo_message "Binding local repository path ${BIND_SRC}"
33:			mkdir -p "${BIND_DEST}"
34:			mount --bind "${BIND_SRC}" "${BIND_DEST}"
35:		fi
36:	fi
37:}
38:
39:Chroot_try_unbind_path ()
40:{
41:	local CHROOT
42:	local BIND_SRC
43:	local BIND_DEST
44:	CHROOT="$(readlink -f ${1})"
45:	if echo "${2}" | grep -E -q '^file://(localhost)?/'
46:	then
47:		BIND_SRC="$(readlink -f $(echo ${2} | sed --regexp-extended -e 's|^file://(localhost)?||'))"
48:
49:		BIND_DEST=${CHROOT}${BIND_SRC}
50:		if [ -d "${BIND_DEST}" ]
51:		then
52:			Echo_message "Unbinding local repository path"
53:			umount "${BIND_DEST}"  > /dev/null 2>&1 || true
54:			rmdir --parents "${BIND_DEST}" || true
55:		fi
56:	fi
57:}
